Simulate One Generation

# FUNCTION 1: SIMULATE ONE GENERATION

# The model includes two turtle types:
# a = usual turtles, which are affected by warming
# A = adapted turtles, which have a relative growth advantage

simulate_one_gen <- function(N_a, N_A, decay_rate, sel_coeff, mut_rate, mig_rate, N_source_a, carrying_capacity = 1000) {

# STEP 1: Generate offspring for each turtle type

# For type a, the expected number of offspring is reduced
# by the warming-related decay rate.
# For type A, offspring production depends on its current population, its selection advantage, warming and how close the total population is to carrying capacity.
  offsp_a <- rpois(1, N_a * (1 - decay_rate))
  offsp_A <- rpois(1, N_A + N_A * (sel_coeff - decay_rate) * (1 - (N_a + N_A) / carrying_capacity))

# Calculate the total number of offspring of both types.
  offsp_total <- offsp_A + offsp_a

# If the total exceeds carrying capacity, randomly retain individuals until the population is within the limit.
  if (offsp_total > carrying_capacity) 
    {
    
    offsp_a <- rbinom(1, carrying_capacity, offsp_a / offsp_total)

# The remaining individuals belong to type A.
    offsp_A <- carrying_capacity - offsp_a
  }

# STEP 2: Simulate mutation from type a to type A

# The expected number of mutations depends on the number of type-a offspring and the mutation rate.
  mut_a_to_A <- rpois(1, offsp_a * mut_rate)

# STEP 3: Simulate immigration from the source beach

# The number of immigrants is determined by the source population size and the migration rate.
# In this model, immigrants are type a, not type A.
  mig_a <- rpois(1, N_source_a * mig_rate)

# STEP 4: Calculate the population sizes in the next generation

# Type a loses individuals through mutation but gains individuals through immigration.
  N_a_new <- offsp_a - mut_a_to_A + mig_a

# Type A gains the individuals that mutated from type a.
# Existing type-A offspring remain part of this population.
  N_A_new <- offsp_A + mut_a_to_A

# Return the updated populations for the next generation.
  return(c(
    N_a = N_a_new,
    N_A = N_A_new
  ))

}

# FUNCTION 2: SIMULATE THE POPULATION

# This function simulates how the populations of type a and type A change over several generations.

# It uses simulate_one_gen() to calculate the population in the next generation and saves the results at every step.

simulate_pop <- function(N_init_a, N_init_A, decay_rate, sel_coeff,
                         mut_rate, mig_rate, N_source_a, t_max,
                         carrying_capacity = 1000) {

# Store the initial population.
# The simulation starts with N_init_a usual turtles and N_init_A adapted turtles.
  pop_vector <- c(N_init_a, N_init_A)

# Store the current population.
# This vector is updated at every generation.
  pop_new <- c(N_init_a, N_init_A)

# Run the simulation for t_max generations.
# seq_len(t_max) produces the numbers from 1 to t_max.
  for (i in seq_len(t_max)) {

# Simulate the next generation using Function 1.
# pop_new[1] is the current number of a turtles.
# pop_new[2] is the current number of A turtles.
# The other describe warming, selection, mutation, migration and carrying capacity.
    pop_new <- simulate_one_gen(
      pop_new[1], pop_new[2],
      decay_rate, sel_coeff, mut_rate,
      mig_rate, N_source_a, carrying_capacity
    )

# Add the newly calculated population as a new row.
# This stores the population sizes for every generation, rather than keeping only the final result.
    pop_vector <- rbind(pop_vector, pop_new)
  }

# Label the rows with their generation numbers.
# The first row is generation 0, which contains the initial population. The last row corresponds to generation t_max.
  rownames(pop_vector) <- 0:(nrow(pop_vector) - 1)

# Name the two columns according to the turtle types.
# Generations remain row names; they are not a population column.
  colnames(pop_vector) <- c("a", "A")

# Return the complete population trajectory.
# The result contains two columns (a and A) and one row for the initial population and every simulated generation.
  return(pop_vector)
}

# Migration rates to compare
migration_rates <- seq(0, 0.10, by = 0.01)

# Repeat simulations to account for random variation
n_replicates <- 30

# Store the mean results for each migration rate
adapted_A <- numeric(length(migration_rates))
mig_a <- 1000 * migration_rates # Expected number of migrating turtles per generation
population_size <- numeric(length(migration_rates))


for (j in seq_along(migration_rates)) {

  # Store the final results of repeated simulations
  results <- matrix(NA_real_, nrow = n_replicates, ncol = 3)

  for (rep in seq_len(n_replicates)) {

    sim <- simulate_pop(
      N_init_a = 1000,
      N_init_A = 0,
      decay_rate = 0.1,
      sel_coeff = 0.2,
      mut_rate = 0.001,
      mig_rate = migration_rates[j], 
      N_source_a = 1000,
      t_max = max_gen,
      carrying_capacity = 1000
    )

# Number of A and a turtles at the final generation
    results[rep, 1] <- sim[nrow(sim), "A"]
    
    results[rep, 2] <- sim[nrow(sim), "a"]
    
# Total population at the final generation
    results[rep, 3] <- sum(sim[nrow(sim), ])
  }

# Calculate the mean across replicate simulations
  adapted_A[j] <- mean(results[, 1])
  mig_a [j] <- mean(results[, 2])
  population_size[j] <- mean(results[, 3])
}
